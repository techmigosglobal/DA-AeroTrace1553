cmake_minimum_required(VERSION 3.24)
project(AeroTrace1553 VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AEROTRACE_BUILD_GUI "Build the native desktop application" ON)
option(AEROTRACE_SANITIZERS "Instrument the backend with ASan and UBSan" OFF)
option(AEROTRACE_BUILD_FUZZERS "Build Clang libFuzzer targets" OFF)
option(AEROTRACE_ENABLE_DDC "Build providers using a locally installed SDK" ON)
if(AEROTRACE_SANITIZERS OR AEROTRACE_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "This native verification profile requires Clang")
  endif()
  if(AEROTRACE_BUILD_GUI)
    message(FATAL_ERROR "Use the separate backend verification build; keep Qt on its matching compiler kit")
  endif()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
  if(MSVC)
    add_link_options(/INCREMENTAL:NO /OPT:NOICF)
    get_filename_component(clang_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    execute_process(COMMAND "${clang_bin}/clang.exe" --print-resource-dir OUTPUT_VARIABLE clang_resource OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    set(AEROTRACE_SANITIZER_LIBDIR "${clang_resource}/lib/windows")
    link_libraries("${AEROTRACE_SANITIZER_LIBDIR}/clang_rt.asan_dynamic-x86_64.lib")
    add_link_options("/wholearchive:${AEROTRACE_SANITIZER_LIBDIR}/clang_rt.asan_static_runtime_thunk-x86_64.lib")
  else()
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()

set(AEROTRACE_DOCUMENT_PYTHON "" CACHE FILEPATH "Python with AeroTrace document dependencies")
set(document_python_candidates "${AEROTRACE_DOCUMENT_PYTHON}"
  "${PROJECT_SOURCE_DIR}/out/document-runtime/Scripts/python.exe"
  "${PROJECT_SOURCE_DIR}/out/document-runtime-local/Scripts/python.exe"
  "$ENV{ProgramFiles}/AeroTrace1553/document-runtime/python.exe"
  "$ENV{LOCALAPPDATA}/Programs/AeroTrace1553/document-runtime/python.exe")
set(AEROTRACE_DOCUMENT_PYTHON "")
foreach(candidate IN LISTS document_python_candidates)
  if(EXISTS "${candidate}")
    execute_process(COMMAND ${CMAKE_COMMAND} -E env OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1
      "${candidate}" -c "import numpy, onnxruntime, pdfplumber"
      RESULT_VARIABLE python_healthy OUTPUT_QUIET ERROR_QUIET TIMEOUT 60)
    if(python_healthy EQUAL 0)
      set(AEROTRACE_DOCUMENT_PYTHON "${candidate}")
      set(AEROTRACE_DOCUMENT_PYTHON "${candidate}" CACHE FILEPATH "Python with AeroTrace document dependencies" FORCE)
      break()
    else()
      message(WARNING "Document runtime health check failed for ${candidate}: ${python_healthy}; document tests are unavailable")
    endif()
  endif()
endforeach()
