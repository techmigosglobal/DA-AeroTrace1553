cmake_minimum_required(VERSION 3.24)
project(AeroTrace1553 VERSION 0.5.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AEROTRACE_BUILD_GUI "Build the native desktop application" ON)
enable_testing()
if(WIN32)
  add_test(NAME support_bundle_collection COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${PROJECT_SOURCE_DIR}/scripts/test-diagnostics.ps1" -ReportDirectory "${CMAKE_BINARY_DIR}/test-results/diagnostics")
  set_tests_properties(support_bundle_collection PROPERTIES TIMEOUT 45)
endif()
add_subdirectory(modules/core)
add_subdirectory(modules/acquisition)
add_subdirectory(modules/application)
add_executable(aerotrace_core_cli tools/core_cli/main.cpp)
target_link_libraries(aerotrace_core_cli PRIVATE aerotrace_core)
add_executable(aerotrace_hardware_probe tools/core_cli/hardware_probe.cpp)
target_link_libraries(aerotrace_hardware_probe PRIVATE aerotrace_acquisition)
add_executable(aerotrace_qualification tools/core_cli/qualification.cpp)
target_link_libraries(aerotrace_qualification PRIVATE aerotrace_acquisition)
add_test(NAME qualification_cli_help COMMAND aerotrace_qualification --help)
if(TARGET aerotrace_ddc_test_runtime)
  add_test(NAME qualification_four_stream_fixture COMMAND ${CMAKE_COMMAND}
    "-DQUALIFIER=$<TARGET_FILE:aerotrace_qualification>" "-DRUNTIME=$<TARGET_FILE:aerotrace_ddc_test_runtime>"
    "-DOUTPUT_ROOT=${CMAKE_BINARY_DIR}" -P "${PROJECT_SOURCE_DIR}/tools/core_cli/qualification_fixture.cmake")
  set_tests_properties(qualification_four_stream_fixture PROPERTIES TIMEOUT 30)
endif()
option(AEROTRACE_RELEASE_IPO "Enable supported Release interprocedural optimization" ON)
if(AEROTRACE_RELEASE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT aerotrace_ipo_supported OUTPUT aerotrace_ipo_error)
  if(aerotrace_ipo_supported)
    set_property(TARGET aerotrace_core aerotrace_acquisition aerotrace_application
      PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
endif()
set(AEROTRACE_DOCUMENT_PYTHON "" CACHE FILEPATH "Python with AeroTrace document dependencies")
set(document_python_candidates "${AEROTRACE_DOCUMENT_PYTHON}"
  "${PROJECT_SOURCE_DIR}/out/document-runtime/Scripts/python.exe"
  "${PROJECT_SOURCE_DIR}/out/document-runtime-local/Scripts/python.exe"
  "$ENV{ProgramFiles}/AeroTrace1553/document-runtime/python.exe"
  "$ENV{LOCALAPPDATA}/Programs/AeroTrace1553/document-runtime/python.exe")
set(AEROTRACE_DOCUMENT_PYTHON "")
foreach(candidate IN LISTS document_python_candidates)
  if(EXISTS "${candidate}")
    execute_process(COMMAND "${candidate}" -c "import numpy, onnxruntime, pdfplumber"
      RESULT_VARIABLE python_healthy OUTPUT_QUIET ERROR_QUIET TIMEOUT 15)
    if(python_healthy EQUAL 0)
      set(AEROTRACE_DOCUMENT_PYTHON "${candidate}")
      break()
    endif()
  endif()
endforeach()
if(AEROTRACE_BUILD_GUI)
  add_subdirectory(apps/desktop)
endif()
if(AEROTRACE_DOCUMENT_PYTHON)
  message(STATUS "Document runtime: ${AEROTRACE_DOCUMENT_PYTHON}")
  add_test(NAME offline_pdf_ocr_and_retrieval
    COMMAND "${AEROTRACE_DOCUMENT_PYTHON}" "${PROJECT_SOURCE_DIR}/tools/document_worker/tests.py")
  set_tests_properties(offline_pdf_ocr_and_retrieval PROPERTIES TIMEOUT 120 ENVIRONMENT "PYTHONUTF8=1")
  add_test(NAME icd_pdf_fixture COMMAND "${AEROTRACE_DOCUMENT_PYTHON}"
    "${PROJECT_SOURCE_DIR}/tools/document_worker/tests.py" --write-fixture "${CMAKE_BINARY_DIR}/test-results/icd-fixture.pdf")
  set_tests_properties(icd_pdf_fixture PROPERTIES FIXTURES_SETUP icd_pdf)
  if(AEROTRACE_BUILD_GUI)
    add_test(NAME gui_pdf_to_live_conversion COMMAND aerotrace_desktop --document-workflow-test
      "${CMAKE_BINARY_DIR}/test-results" "${CMAKE_BINARY_DIR}/test-results/icd-fixture.pdf")
    set_tests_properties(gui_pdf_to_live_conversion PROPERTIES FIXTURES_REQUIRED icd_pdf ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 150)
  endif()
endif()
