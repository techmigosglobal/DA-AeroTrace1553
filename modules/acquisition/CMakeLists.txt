  set(provider_header_files "")
  foreach(include_dir IN LISTS DDC_INCLUDE_DIRS)
    file(GLOB_RECURSE include_headers "${include_dir}/*.h")
    list(APPEND provider_header_files ${include_headers})
  endforeach()
  list(REMOVE_DUPLICATES provider_header_files)
  list(SORT provider_header_files)
  set(provider_header_manifest "")
  foreach(header IN LISTS provider_header_files)
    file(SHA256 "${header}" header_hash)
    file(RELATIVE_PATH relative_header "${DDC_SDK_ROOT}" "${header}")
    string(APPEND provider_header_manifest "${relative_header}:${header_hash}\n")
  endforeach()
  string(SHA256 provider_header_hash "${provider_header_manifest}")
  get_filename_component(provider_sdk_name "${DDC_SDK_ROOT}" NAME)
  if(AEROTRACE_DDC_SDK_IDENTITY)
    set(provider_sdk_name "${AEROTRACE_DDC_SDK_IDENTITY}")

      AEROTRACE_PROVIDER_ID="${AEROTRACE_DDC_PROVIDER_ID}" AEROTRACE_PROVIDER_SDK="${provider_sdk_name}"
      AEROTRACE_PROVIDER_HEADER_HASH="${provider_header_hash}" AEROTRACE_PROVIDER_RUNTIME_HASH="${provider_runtime_hash}")
    target_link_libraries(${provider_target} PRIVATE setupapi cfgmgr32 version advapi32 bcrypt)
  endforeach()
