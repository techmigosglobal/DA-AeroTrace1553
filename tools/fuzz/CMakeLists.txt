if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Fuzzing requires Clang with libFuzzer")
endif()
add_library(aerotrace_fuzz_core STATIC ../../modules/core/core.cpp ../../modules/core/protocol.cpp)
target_include_directories(aerotrace_fuzz_core PUBLIC "${PROJECT_SOURCE_DIR}/modules/core")
target_compile_options(aerotrace_fuzz_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
if(MSVC)
  # Match the prebuilt LLVM libFuzzer STL annotation ABI. Backend sanitizer tests
  # retain the MSVC container annotations; only fuzz binaries use this setting.
  target_compile_definitions(aerotrace_fuzz_core PUBLIC _DISABLE_STRING_ANNOTATION _DISABLE_VECTOR_ANNOTATION)
endif()
foreach(name recording protocol packet)
  add_executable(aerotrace_fuzz_${name} ${name}_fuzz.cpp)
  target_link_libraries(aerotrace_fuzz_${name} PRIVATE aerotrace_fuzz_core)
  target_compile_options(aerotrace_fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
  if(NOT MSVC)
    target_link_options(aerotrace_fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
  else()
    target_link_libraries(aerotrace_fuzz_${name} PRIVATE "${AEROTRACE_SANITIZER_LIBDIR}/clang_rt.fuzzer-x86_64.lib")
  endif()
endforeach()
add_executable(aerotrace_sanitizer_canary sanitizer_canary.cpp)
add_test(NAME sanitizer_runtime_canaries COMMAND ${CMAKE_COMMAND} "-DCANARY=$<TARGET_FILE:aerotrace_sanitizer_canary>" -P "${CMAKE_CURRENT_SOURCE_DIR}/check_canary.cmake")
