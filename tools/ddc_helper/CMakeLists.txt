cmake_minimum_required(VERSION 3.24)
project(AeroTraceDdcHelper VERSION 0.6.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build the helper separately with MSVC x86; do not change the desktop Qt kit")
endif()
set(DDC_SDK_ROOT "C:/DDC/aceXtremeSDKv3.6.15" CACHE PATH "Licensed SDK matching the helper runtime")
if(NOT EXISTS "${DDC_SDK_ROOT}/Include/stdemace.h" OR NOT EXISTS "${DDC_SDK_ROOT}/Lib/Win32/Release/emacepl.dll")
  message(FATAL_ERROR "Matching SDK headers and Win32 runtime are required")
endif()
get_filename_component(AEROTRACE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ACQUISITION "${AEROTRACE_ROOT}/modules/acquisition")
set(CORE "${AEROTRACE_ROOT}/modules/core")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/EHsc /W3)
include_directories("${ACQUISITION}" "${CORE}" "${DDC_SDK_ROOT}/Include")
add_executable(aerotrace_ddc_helper_x86 main.cpp support.cpp "${ACQUISITION}/helper_packs.cpp" "${ACQUISITION}/providers/ddc_vendor_source.cpp")
target_compile_definitions(aerotrace_ddc_helper_x86 PRIVATE AEROTRACE_DDC=1)
target_link_libraries(aerotrace_ddc_helper_x86 PRIVATE bcrypt advapi32 setupapi cfgmgr32)
add_executable(aerotrace_helper_protocol_tests "${ACQUISITION}/helper_protocol_tests.cpp")
add_executable(aerotrace_helper_integration_tests "${ACQUISITION}/helper_integration_tests.cpp" "${ACQUISITION}/helper_source.cpp" "${ACQUISITION}/helper_packs.cpp" "${CORE}/core.cpp" "${CORE}/protocol.cpp")
target_link_libraries(aerotrace_helper_integration_tests PRIVATE bcrypt advapi32 setupapi cfgmgr32)
enable_testing()
add_test(NAME helper_wire_contract_x86 COMMAND aerotrace_helper_protocol_tests)
